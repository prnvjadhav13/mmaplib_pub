cmake_minimum_required(VERSION 3.20)
project(linuxfs_mmap LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_library(linuxfs_mmap mmap.cpp)
else()
    add_library(linuxfs_mmap mmap_unsupported.cpp)
endif()
target_include_directories(linuxfs_mmap PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
    target_compile_options(linuxfs_mmap PRIVATE /W4 /permissive-)
else()
    target_compile_options(linuxfs_mmap PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()

include(CTest)
if(BUILD_TESTING)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        add_executable(linuxfs_mmap_tests mmap_test.cpp)
        target_link_libraries(linuxfs_mmap_tests PRIVATE linuxfs_mmap)
        add_test(NAME linuxfs_mmap_tests COMMAND linuxfs_mmap_tests)
    endif()
endif()

install(TARGETS linuxfs_mmap ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(FILES mmap.hpp DESTINATION include/linuxfs)
